  // Si hay archivo seleccionado
  if (archivoInput.current.files[0]) {
    formData.append('documento', archivoInput.current.files[0]);
  }
  
  try {
    const response = await fetch('http://localhost:3000/api/citas', {
      method: 'POST',
      body: formData
    });
    
    const data = await response.json();
    
    if (data.success) {
      console.log('Cita creada:', data.data);
      
      // Si se subió documento, la URL estará disponible
      if (data.data.documento) {
        console.log('Documento URL:', data.data.documento.enlace);
      }
    } else {
      console.error('Error:', data.error);
    }
  } catch (error) {
    console.error('Error al crear cita:', error);
  }
};

5. COMPONENTE REACT COMPLETO
================================

import React, { useState, useRef } from 'react';
import axios from 'axios';

function CrearCita() {
  const [formData, setFormData] = useState({
    id_paciente: '',
    id_medico: '',
    fecha_cita: '',
    motivo: '',
    tipo_documento: 'Documento de cita'
  });
  const [archivo, setArchivo] = useState(null);
  const fileInputRef = useRef(null);

  const handleChange = (e) => {
    setFormData({
      ...formData,
      [e.target.name]: e.target.value
    });
  };

  const handleFileChange = (e) => {
    setArchivo(e.target.files[0]);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();

    const data = new FormData();
    data.append('id_paciente', formData.id_paciente);
    data.append('id_medico', formData.id_medico);
    data.append('fecha_cita', formData.fecha_cita);
    data.append('motivo', formData.motivo);
    
    if (archivo) {
      data.append('documento', archivo);
      data.append('tipo_documento', formData.tipo_documento);
    }

    try {
      const response = await axios.post('http://localhost:3000/api/citas', data, {
        headers: {
          'Content-Type': 'multipart/form-data'
        }
      });

      alert('Cita creada exitosamente');
      console.log('Respuesta:', response.data);

      // Limpiar formulario
      setFormData({
        id_paciente: '',
        id_medico: '',
        fecha_cita: '',
        motivo: '',
        tipo_documento: 'Documento de cita'
      });
      setArchivo(null);
      if (fileInputRef.current) {
        fileInputRef.current.value = '';
      }
    } catch (error) {
      console.error('Error:', error);
      alert('Error al crear la cita: ' + (error.response?.data?.error || error.message));
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <label>ID Paciente:</label>
        <input
          type="text"
          name="id_paciente"
          value={formData.id_paciente}
          onChange={handleChange}
          required
        />
      </div>

      <div>
        <label>ID Médico:</label>
        <input
          type="text"
          name="id_medico"
          value={formData.id_medico}
          onChange={handleChange}
          required
        />
      </div>

      <div>
        <label>Fecha de la Cita:</label>
        <input
          type="datetime-local"
          name="fecha_cita"
          value={formData.fecha_cita}
          onChange={handleChange}
          required
        />
      </div>

      <div>
        <label>Motivo:</label>
        <textarea
          name="motivo"
          value={formData.motivo}
          onChange={handleChange}
        />
      </div>

      <div>
        <label>Tipo de Documento:</label>
        <input
          type="text"
          name="tipo_documento"
          value={formData.tipo_documento}
          onChange={handleChange}
        />
      </div>

      <div>
        <label>Documento (opcional):</label>
        <input
          type="file"
          ref={fileInputRef}
          onChange={handleFileChange}
          accept=".pdf,.jpg,.jpeg,.png,.doc,.docx,.xls,.xlsx"
        />
        {archivo && <p>Archivo seleccionado: {archivo.name}</p>}
      </div>

      <button type="submit">Crear Cita</button>
    </form>
  );
}

export default CrearCita;

================================
7. VERIFICAR DOCUMENTO SUBIDO
================================

Una vez creada la cita con documento, puedes:

1. Acceder a la URL del documento directamente desde el campo "enlace"
2. Ver el documento en Supabase Dashboard:
   - Ve a Storage > documentos-citas
   - Navega a la carpeta "documentos/"
   - Verás el archivo subido con timestamp

3. Consultar la cita con su documento:

curl http://localhost:3000/api/citas/{id_cita}

================================
8. ERRORES COMUNES Y SOLUCIONES
================================

ERROR: "El archivo es demasiado grande"
SOLUCIÓN: El archivo no debe superar 10MB

ERROR: "Tipo de archivo no permitido"
SOLUCIÓN: Solo se permiten: PDF, JPG, PNG, DOC, DOCX, XLS, XLSX

ERROR: "Error al subir el archivo: The resource was not found"
SOLUCIÓN: Verifica que el bucket 'documentos-citas' exista en Supabase

ERROR: "SUPABASE_URL y SUPABASE_SERVICE_KEY no están configurados"
SOLUCIÓN: Configura estas variables en tu archivo .env

================================
9. PASO A PASO PARA PRIMER USO
================================

1. Configura el archivo .env con:
   SUPABASE_URL=https://[tu-proyecto].supabase.co
   SUPABASE_SERVICE_KEY=tu_service_role_key

2. Crea el bucket 'documentos-citas' en Supabase Storage

3. Configura las políticas RLS (ver CONFIGURACION_STORAGE_CITAS.md)

4. Reinicia el servidor backend:
   npm run dev

5. Obtén IDs válidos de paciente y médico:
   curl http://localhost:3000/api/personas
   curl http://localhost:3000/api/medicos

6. Prueba crear una cita sin documento primero

7. Luego prueba con un documento PDF pequeño
